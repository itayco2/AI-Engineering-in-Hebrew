# This file is deliberately minimal. It has no torch, no sentence-transformers and no
# jupyter, because the whole point of keeping logic out of notebooks is that the logic can
# be tested without any of that. Adding a heavy dependency here turns a 40-second CI run
# into a 6-minute one, and a slow badge is a badge nobody trusts.

pydantic==2.12.3

# numpy drops old Python quickly: 2.5 needs 3.12, and 2.4 needs 3.11. Pinned per interpreter
# rather than loosened, so every version in the advertised support matrix actually resolves.
# The rule this enforces: the matrix we claim is the matrix we test.
numpy==2.5.3; python_version >= "3.12"
numpy==2.4.6; python_version == "3.11"
numpy==2.2.6; python_version < "3.11"
